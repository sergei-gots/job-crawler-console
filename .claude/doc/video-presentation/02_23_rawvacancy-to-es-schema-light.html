5/**
6 * Upserts by a deterministic id (`sourceId:externalId`) so a vacancy re-seen on a later crawl —
7 * possibly by a different user's crawler job — updates `lastSeenAt` instead of duplicating.
8 * `firstSeenAt` is only set on the initial insert. `listingId` (Increment 9) is best-effort,
9 * not part of the dedup key — see `CRAWLER_RESULTS_SCHEMA_VERSION`'s v4 note: a vacancy that
10 * appears in more than one of a source's listings ends up attributed to whichever crawled it
11 * last, since the underlying record is genuinely one shared vacancy, not one per listing.
12 */
13export async function upsertVacancy(raw: RawVacancy, listingId: number | null = null): Promise<void> {
14  await ensureCrawlerResultsIndex();
15 
16  const id = `${raw.sourceId}:${raw.externalId}`;
17  const now = new Date().toISOString();
18 
19  // Detail fields are only included when present on `raw`, so a listing-only upsert never
20  // clobbers detail data written by a later enrichDetails pass (and vice versa).
21  const detailFields = {
22    ...(raw.description !== undefined ? { description: raw.description } : {}),
23    ...(raw.location !== undefined ? { location: raw.location } : {}),
24    ...(raw.isRemote !== undefined ? { isRemote: raw.isRemote } : {}),
25    ...(raw.skillsSummary !== undefined ? { skillsSummary: raw.skillsSummary } : {}),
26    ...(raw.specialization !== undefined ? { specialization: raw.specialization } : {}),
27    ...(raw.seniority !== undefined ? { seniority: raw.seniority } : {}),
28  };
29 
30  await esClient.update({
31    index: CRAWLER_RESULTS_INDEX,
32    id,
33    doc: {
34      sourceId: raw.sourceId,
35      externalId: raw.externalId,
36      title: raw.title,
37      company: raw.company,
38      url: raw.url,
39      postedAt: raw.postedAt,
40      lastSeenAt: now,
41      listingId,
42      ...detailFields,
43    },
44    upsert: {
45      sourceId: raw.sourceId,
46      externalId: raw.externalId,
47      title: raw.title,
48      company: raw.company,
49      url: raw.url,
50      postedAt: raw.postedAt,
51      firstSeenAt: now,
52      lastSeenAt: now,
53      listingId,
54      ...detailFields,
55    },
56  });
57}
