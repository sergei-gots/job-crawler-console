129    const pubDate = item.find("pubDate").first().text().trim();
130 
131    bySlug.set(slug, {

133      // <region> (e.g. "Anywhere in the World", "USA Only") is always populated, unlike the
134      // JobPosting JSON-LD approach this replaced, where jobLocation was only present for
135      // geo-restricted postings - a strictly more complete signal for the same field.
136      location: region,
137      // WWR is a 100%-remote job board by design (same rationale as RemoteOK's hardcoded true).
